# ==== WSL auto proxy (mirrored network, proxy on Windows at 127.0.0.1:7892) ====
# 终端一开就探测 Windows 上的代理端口，通了就自动 export。
# 端口不同就改这里的 7892。
if timeout 1 bash -c "echo > /dev/tcp/127.0.0.1/7892" 2>/dev/null; then
    export http_proxy="http://127.0.0.1:7892"
    export https_proxy="http://127.0.0.1:7892"
    export all_proxy="http://127.0.0.1:7892"
    export no_proxy="localhost,127.0.0.1"
fi
proxy_off() { unset http_proxy https_proxy all_proxy; }
proxy_on() {
    export http_proxy="http://127.0.0.1:7892" https_proxy="http://127.0.0.1:7892" all_proxy="http://127.0.0.1:7892" no_proxy="localhost,127.0.0.1"
}
# ==== end auto proxy ====
